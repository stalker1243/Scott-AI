// Логика лаунчера: заполнение начальных данных, обработка колбэков из UI
// и переключение темы Classic / Glass (в т.ч. нативный blur окна там,
// где это поддерживается ОС).

slint::include_modules!();

use slint::{Color, ComponentHandle, Model, ModelRc, SharedString, VecModel};

/// Режим отображения окна — влияет и на цвета (Palette в .slint),
/// и на нативный blur/acrylic окна.
#[derive(Clone, Copy, PartialEq, Eq)]
enum ThemeMode {
    Classic,
    Glass,
}

fn main() -> Result<(), slint::PlatformError> {
    let app = AppWindow::new()?;

    // ---------- начальные данные ----------
    seed_chat_history(&app);
    seed_chat_messages(&app);

    // Стартовая тема — Classic (соответствует Palette.glass-mode: false по умолчанию).
    apply_theme(&app, ThemeMode::Classic);

    // ---------- колбэки: вкладка "Основное" ----------
    {
        let app_weak = app.as_weak();
        app.on_launch_assistant(move || {
            let Some(app) = app_weak.upgrade() else { return };
            // Здесь должен быть реальный запуск ассистента (спавн процесса,
            // подключение к бэкенду и т.п.). Пока просто переключаемся на чат.
            app.set_current_tab(1);
        });
    }

    // ---------- колбэки: вкладка "Чат" ----------
    {
        let app_weak = app.as_weak();
        app.on_new_chat(move || {
            let Some(app) = app_weak.upgrade() else { return };
            let messages = ModelRc::new(VecModel::from(Vec::<ChatMessage>::new()));
            app.set_chat_messages(messages);
        });
    }

    {
        let app_weak = app.as_weak();
        app.on_history_item_selected(move |index| {
            let Some(app) = app_weak.upgrade() else { return };
            let history = app.get_chat_history();
            let model = history
                .as_any()
                .downcast_ref::<VecModel<ChatHistoryItem>>()
                .expect("chat-history должен быть VecModel");

            for i in 0..model.row_count() {
                if let Some(mut item) = model.row_data(i) {
                    item.active = i as i32 == index;
                    model.set_row_data(i, item);
                }
            }
            // TODO: подгрузить реальные сообщения выбранного диалога.
        });
    }

    {
        let app_weak = app.as_weak();
        app.on_send_message(move || {
            let Some(app) = app_weak.upgrade() else { return };
            let text = app.get_chat_draft();
            if text.trim().is_empty() {
                return;
            }

            let messages = app.get_chat_messages();
            let model = messages
                .as_any()
                .downcast_ref::<VecModel<ChatMessage>>()
                .expect("chat-messages должен быть VecModel");

            model.push(ChatMessage {
                from_user: true,
                text: text.clone(),
                time: current_time(),
            });
            app.set_chat_draft(SharedString::from(""));

            // Заглушка ответа ассистента — здесь позже будет реальный вызов backend/AI.
            model.push(ChatMessage {
                from_user: false,
                text: "Принял ваше сообщение! (это демо-ответ — подключите сюда реальный backend)"
                    .into(),
                time: current_time(),
            });
        });
    }

    // ---------- колбэки: "Настройки" ----------
    {
        let app_weak = app.as_weak();
        app.on_style_changed(move |is_glass| {
            let Some(app) = app_weak.upgrade() else { return };
            apply_theme(&app, if is_glass { ThemeMode::Glass } else { ThemeMode::Classic });
        });
    }

    {
        let app_weak = app.as_weak();
        app.on_accent_changed(move |color: Color| {
            let Some(app) = app_weak.upgrade() else { return };
            app.global::<Theme>().set_accent(color);
        });
    }

    // ---------- колбэки: "Профиль" ----------
    app.on_save_changes(move || {
        // TODO: сохранить профиль/настройки на диск (например, в JSON рядом с exe).
        println!("Изменения профиля и настроек сохранены (заглушка).");
    });

    app.run()
}

/// Применяет тему: переключает Palette.glass-mode и, где возможно,
/// добавляет/убирает нативный blur самого окна ОС.
fn apply_theme(app: &AppWindow, mode: ThemeMode) {
    let is_glass = mode == ThemeMode::Glass;
    app.global::<Theme>().set_glass(is_glass);

    #[cfg(target_os = "windows")]
    {
        let handle = app.window().window_handle();
        if is_glass {
            let _ = window_vibrancy::apply_acrylic(&handle, Some((16, 12, 28, 160)));
        } else {
            let _ = window_vibrancy::clear_acrylic(&handle);
        }
    }

    #[cfg(target_os = "macos")]
    {
        let handle = app.window().window_handle();
        if is_glass {
            let _ = window_vibrancy::apply_vibrancy(
                &handle,
                window_vibrancy::NSVisualEffectMaterial::HudWindow,
                None,
                None,
            );
        } else {
            let _ = window_vibrancy::clear_vibrancy(&handle);
        }
    }

    #[cfg(not(any(target_os = "windows", target_os = "macos")))]
    {
        let _ = is_glass;
    }
}

fn seed_chat_history(app: &AppWindow) {
    let items = vec![
        ChatHistoryItem { icon: "❓".into(), title: "Что такое квантовая...".into(), time: "10:45".into(), active: true },
        ChatHistoryItem { icon: "</>".into(), title: "Помоги с кодом".into(), time: "09:30".into(), active: false },
        ChatHistoryItem { icon: "🖼".into(), title: "Анализ изображения".into(), time: "Вчера".into(), active: false },
        ChatHistoryItem { icon: "💡".into(), title: "Идеи для проекта".into(), time: "Вчера".into(), active: false },
        ChatHistoryItem { icon: "📖".into(), title: "Объясни теорию...".into(), time: "Вчера".into(), active: false },
    ];
    app.set_chat_history(ModelRc::new(VecModel::from(items)));
}

fn seed_chat_messages(app: &AppWindow) {
    let messages = vec![
        ChatMessage { from_user: true, text: "Привет! Расскажи, что ты можешь делать?".into(), time: "10:45".into() },
        ChatMessage {
            from_user: false,
            text: "Привет! Я могу помочь вам с самыми разными задачами: отвечать на вопросы, \
                   анализировать данные, писать код, объяснять сложные темы, работать с \
                   изображениями и многое другое. Просто спросите!"
                .into(),
            time: "10:45".into(),
        },
    ];
    app.set_chat_messages(ModelRc::new(VecModel::from(messages)));
}

fn current_time() -> SharedString {
    "сейчас".into()
}
