cmake_minimum_required(VERSION 3.24)
project(ScottSystem LANGUAGES CXX)
include(CTest)
if(WIN32)
    add_executable(ScottSystem SystemControl.cpp)
    target_compile_features(ScottSystem PRIVATE cxx_std_17)
    target_link_libraries(ScottSystem PRIVATE ole32 uuid)
    if(MINGW)
        target_link_options(ScottSystem PRIVATE -static -s)
    endif()
    if(BUILD_TESTING)
        add_test(NAME native_system_contract COMMAND ${CMAKE_COMMAND}
            -DNATIVE=$<TARGET_FILE:ScottSystem> -P ${CMAKE_CURRENT_SOURCE_DIR}/verify_cli.cmake)
    endif()
endif()
