cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION.json" SCOTT_VERSION_JSON)
string(JSON SCOTT_VERSION GET "${SCOTT_VERSION_JSON}" version)
string(JSON SCOTT_BUILD GET "${SCOTT_VERSION_JSON}" build_number)
project(ScottAIQt VERSION ${SCOTT_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Widgets Test)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_library(scott_client STATIC src/BackendClient.cpp src/BackendClient.h src/ProtocolClient.cpp src/ProtocolValidation.h src/ModelClient.cpp src/ChatClient.cpp)
target_link_libraries(scott_client PUBLIC Qt6::Network Qt6::Widgets)
target_include_directories(scott_client PUBLIC src)
qt_add_library(scott_tray STATIC src/TrayController.cpp src/TrayController.h)
target_link_libraries(scott_tray PUBLIC scott_client Qt6::Quick Qt6::Widgets)
qt_add_library(scott_avatar STATIC src/ProfileAvatar.cpp src/ProfileAvatar.h)
target_link_libraries(scott_avatar PUBLIC Qt6::Quick Qt6::Widgets)
qt_add_library(scott_protocol_files STATIC src/ProtocolFiles.cpp src/ProtocolFiles.h)
target_link_libraries(scott_protocol_files PUBLIC scott_client Qt6::Widgets)
qt_add_executable(ScottAIQt src/main.cpp src/WindowEffects.cpp src/WindowEffects.h src/WindowMotion.cpp src/WindowMotion.h src/SetupController.cpp src/SetupController.h)
if(WIN32)
    target_sources(ScottAIQt PRIVATE app.rc)
    target_link_libraries(ScottAIQt PRIVATE dwmapi)
endif()
target_link_libraries(ScottAIQt PRIVATE scott_tray scott_avatar scott_protocol_files Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
target_compile_definitions(ScottAIQt PRIVATE SCOTT_BACKEND_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../backend")
configure_file(version.h.in "${CMAKE_CURRENT_BINARY_DIR}/ScottVersion.h" @ONLY)
target_include_directories(ScottAIQt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
qt_add_qml_module(ScottAIQt URI Scott.Prototype VERSION 1.0
    QML_FILES Main.qml MemoryPage.qml NavIcon.qml UiButton.qml FieldSurface.qml UiNotice.qml UiScrollBar.qml
              StyleCatalog.js StyleBackdrop.qml SurfacePanel.qml UiSlider.qml UiComboBox.qml UiSwitch.qml UiSegmentedControl.qml SettingsPage.qml
              WindowTitleBar.qml WindowResizeHandles.qml AppNavigation.qml ProfilePage.qml ActionsPage.qml ProtocolsPage.qml ModelsPage.qml CapabilityBadges.qml CapabilityInfo.js SetupDialog.qml)
qt_add_resources(ScottAIQt branding PREFIX /brand BASE ../ScottAI_avalonia/Assets
    FILES ../ScottAI_avalonia/Assets/scott.ico
          ../ScottAI_avalonia/Assets/scott-light.ico
          ../ScottAI_avalonia/Assets/scott-logo.png
          ../ScottAI_avalonia/Assets/scott-logo-light.png)
qt_add_resources(ScottAIQt preview_fixture PREFIX /test FILES tests/PreviewData.qml tests/GlassFixture.qml)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(client_tests tests/ClientTests.cpp)
    target_link_libraries(client_tests PRIVATE scott_client Qt6::Test)
    add_test(NAME client_contract COMMAND client_tests)
    qt_add_executable(chat_tests tests/ChatTests.cpp)
    target_link_libraries(chat_tests PRIVATE scott_client Qt6::Test)
    add_test(NAME chat_contract COMMAND chat_tests)
    qt_add_executable(setup_probe tests/SetupProbe.cpp)
    target_link_libraries(setup_probe PRIVATE Qt6::Core)
    qt_add_executable(setup_tests tests/SetupTests.cpp src/SetupController.cpp src/SetupController.h)
    target_include_directories(setup_tests PRIVATE src)
    target_link_libraries(setup_tests PRIVATE Qt6::Test)
    add_dependencies(setup_tests setup_probe)
    add_test(NAME setup_contract COMMAND setup_tests)
    qt_add_executable(tray_tests tests/TrayTests.cpp)
    target_link_libraries(tray_tests PRIVATE scott_tray Qt6::Test)
    add_test(NAME tray_contract COMMAND tray_tests)
    qt_add_executable(profile_tests tests/ProfileTests.cpp)
    target_link_libraries(profile_tests PRIVATE scott_client scott_avatar Qt6::Test)
    add_test(NAME profile_contract COMMAND profile_tests)
    qt_add_executable(actions_tests tests/ActionsTests.cpp)
    target_link_libraries(actions_tests PRIVATE scott_client Qt6::Test)
    add_test(NAME actions_contract COMMAND actions_tests)
    qt_add_executable(model_tests tests/ModelTests.cpp)
    target_link_libraries(model_tests PRIVATE scott_client Qt6::Test)
    add_test(NAME models_contract COMMAND model_tests)
    add_test(NAME models_screens COMMAND ScottAIQt --smoke-models)
    add_test(NAME voice_screens COMMAND ScottAIQt --smoke-voice)
    set_tests_properties(voice_screens PROPERTIES TIMEOUT 15)

    set_tests_properties(qml_screens PROPERTIES TIMEOUT 110)
    # These compare native desktop pixels and require a client Windows compositor.
    # Keep them in the default local suite; hosted Windows Server runs portable tests.
    set_tests_properties(window_lifecycle qml_screens PROPERTIES LABELS native_desktop)
