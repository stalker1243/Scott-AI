cmake_minimum_required(VERSION 3.24)
project(ScottWhisperExperiment LANGUAGES C CXX)

set(WHISPER_SOURCE "" CACHE PATH "Pinned whisper.cpp checkout")
if(NOT EXISTS "${WHISPER_SOURCE}/include/whisper.h")
    message(FATAL_ERROR "Set WHISPER_SOURCE to the v1.9.5 checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" upstream)

add_executable(ScottWhisperBench Worker.cpp)
target_compile_features(ScottWhisperBench PRIVATE cxx_std_17)
option(SCOTT_DECODER_TRACE "Diagnostic hooks in a separate whisper.cpp source copy" OFF)
if(SCOTT_DECODER_TRACE)
    if(NOT EXISTS "${WHISPER_SOURCE}/src/scott_trace.h")
        message(FATAL_ERROR "Decoder trace requires the separately instrumented source copy")
    endif()
    file(READ "${WHISPER_SOURCE}/src/scott_trace.h" SCOTT_TRACE_HELPER)
    string(FIND "${SCOTT_TRACE_HELPER}" "scott_whisper_pool_limit" SCOTT_COMPLETION_API)
    if(SCOTT_COMPLETION_API EQUAL -1)
        message(FATAL_ERROR "Export a new diagnostic copy with current instrument_source.py")
    endif()
    target_compile_definitions(whisper PRIVATE SCOTT_DECODER_TRACE)
    target_compile_definitions(ScottWhisperBench PRIVATE SCOTT_DECODER_TRACE)
    add_executable(ScottTraceChecks TraceChecks.cpp)
    target_compile_features(ScottTraceChecks PRIVATE cxx_std_17)
    target_compile_definitions(ScottTraceChecks PRIVATE SCOTT_DECODER_TRACE)
    target_include_directories(ScottTraceChecks PRIVATE "${WHISPER_SOURCE}/examples")
    target_link_libraries(ScottTraceChecks PRIVATE whisper)
    if(MINGW)
        target_link_options(ScottTraceChecks PRIVATE -static -s)
    endif()
endif()
target_include_directories(ScottWhisperBench PRIVATE "${WHISPER_SOURCE}/examples")
target_link_libraries(ScottWhisperBench PRIVATE whisper)
if(WIN32)
    target_link_libraries(ScottWhisperBench PRIVATE shell32)
endif()
if(MINGW)
    target_link_options(ScottWhisperBench PRIVATE -static -s)
endif()

option(SCOTT_PREFIX_PROBE "Offline comparison with shared mel and forced token prefixes" OFF)
if(SCOTT_PREFIX_PROBE)
    if(NOT SCOTT_DECODER_TRACE)
        message(FATAL_ERROR "Prefix probe requires the separate diagnostic source copy")
    endif()
    add_executable(ScottPrefixProbe PrefixProbe.cpp)
    target_compile_features(ScottPrefixProbe PRIVATE cxx_std_17)
    target_include_directories(ScottPrefixProbe PRIVATE "${WHISPER_SOURCE}/examples")
    target_link_libraries(ScottPrefixProbe PRIVATE whisper)
    if(WIN32)
        target_link_libraries(ScottPrefixProbe PRIVATE shell32)
    endif()
    if(MINGW)
        target_link_options(ScottPrefixProbe PRIVATE -static -s)
    endif()
endif()
